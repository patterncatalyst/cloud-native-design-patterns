cmake_minimum_required(VERSION 3.31)
project(gateway LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

add_compile_options(-Wall -Wextra)
# gcc-toolset-15's libstdc++ is newer than the one in ubi-minimal: link it statically.
add_link_options(-static-libstdc++ -static-libgcc)

# Static archives that depend on each other: let the linker rescan the group.
set(CMAKE_CXX_LINK_EXECUTABLE
    "<CMAKE_CXX_COMPILER> <FLAGS> <CMAKE_CXX_LINK_FLAGS> <LINK_FLAGS> <OBJECTS> -o <TARGET> -Wl,--start-group <LINK_LIBRARIES> -Wl,--end-group")

find_package(Drogon CONFIG REQUIRED)
find_package(gRPC CONFIG REQUIRED)
find_package(protobuf CONFIG REQUIRED)
find_package(cppgraphqlgen CONFIG REQUIRED)
find_package(Threads REQUIRED)

# --- gRPC stubs for the inventory service ---
set(PROTO_SRC "${CMAKE_SOURCE_DIR}/../proto/inventory.proto")
set(PROTO_OUT "${CMAKE_BINARY_DIR}/gen")
file(MAKE_DIRECTORY ${PROTO_OUT})

add_custom_command(
    OUTPUT ${PROTO_OUT}/inventory.pb.cc ${PROTO_OUT}/inventory.pb.h
           ${PROTO_OUT}/inventory.grpc.pb.cc ${PROTO_OUT}/inventory.grpc.pb.h
    COMMAND protobuf::protoc
        --proto_path=${CMAKE_SOURCE_DIR}/../proto
        --cpp_out=${PROTO_OUT}
        --grpc_out=${PROTO_OUT}
        --plugin=protoc-gen-grpc=$<TARGET_FILE:gRPC::grpc_cpp_plugin>
        ${PROTO_SRC}
    DEPENDS ${PROTO_SRC} protobuf::protoc gRPC::grpc_cpp_plugin
)

# --- GraphQL schema code, generated by schemagen and committed (see README) ---
file(GLOB GRAPHQL_SOURCES CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/src/graphql/*.cpp)
add_library(gateway_schema STATIC ${GRAPHQL_SOURCES})
target_include_directories(gateway_schema PUBLIC ${CMAKE_SOURCE_DIR}/src/graphql)
target_link_libraries(gateway_schema PUBLIC
    cppgraphqlgen::graphqlservice cppgraphqlgen::graphqlpeg cppgraphqlgen::graphqlresponse)
# Generated code: keep its warnings out of this project's -Wall -Wextra signal.
target_compile_options(gateway_schema PRIVATE -w)

add_executable(gateway
    src/main.cpp
    ${PROTO_OUT}/inventory.pb.cc
    ${PROTO_OUT}/inventory.grpc.pb.cc)

target_include_directories(gateway PRIVATE ${PROTO_OUT} src)
target_link_libraries(gateway PRIVATE
    gateway_schema
    Drogon::Drogon
    gRPC::grpc++
    protobuf::protobuf
    Threads::Threads)

# --- Unit tests (run by ctest inside the image build) ---
enable_testing()
add_executable(graphql_test tests/graphql_test.cpp)
target_include_directories(graphql_test PRIVATE src)
target_link_libraries(graphql_test PRIVATE gateway_schema Drogon::Drogon Threads::Threads)
add_test(NAME graphql COMMAND graphql_test)
